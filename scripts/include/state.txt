// CLEO writes all 1024 of them into cleo\cleo_saves\cs<slot>.sav when the game is saved,
// reads them back when that slot is loaded, and zeroes them on a new game.
// This mod uses slots 900-919; keep new ones inside that range.
// See docs/STATE.md.

    // --- shared variable slots ---
    SV_INITIALIZED = 900    // 0 = new game / never set up, 1 = defaults applied
    SV_BALANCE = 901        // Ray's bank balance in dollars (separate from SA's own cash)
    SV_CORRUPTION = 902     // 0..100, Honor shown to the player = 100 - corruption
    SV_STORY_DAY = 903      // 0 = Wednesday, 1 = Thursday, 2 = Friday, 3+ = after the deadline
    SV_LAST_CLOCK = 904     // game clock (minutes since midnight) seen last, to detect midnight
    SV_DEV_FORCE_OUTCOME = 905  // dev only: 0 = roll dice, 1/2/3 = force the next frisk outcome

    // --- story rules (GDD sections 3.1-3.3) ---
    START_BALANCE = 450
    HOSPITAL_DEBT = 12500
    DEADLINE_DAY = 2        // Friday
    DEADLINE_CLOCK = 600    // 10:00

    CORRUPTION_MAX = 100
    TIER_GREY = 25          // 0-24 clean, 25-49 grey, 50-74 dirty, 75-100 baron
    TIER_DIRTY = 50
    TIER_BARON = 75
end
