// Police mechanics (Phase 3): stop, frisk, surrender, cuff, escort, book.
//
// On foot, next to a pedestrian, press T: the suspect is stopped and the dice decide (GDD 3.4)
//   70% surrenders, 20% runs, 10% draws a pistol.
// Suspect states:
//   HANDS_UP  - hands raised for a moment, then he kneels
//   KNEELING  - waits on his knees; press T next to him to cuff him (released after 60 s if you leave him)
//   FLEEING   - runs; gives up once you stay close to him for 1.5 s
//   HOSTILE   - shoots at you; drops the gun and gives up when his health falls below 50
//   CUFFED    - Miranda rights, evidence found on him (0-3 packages), then he stands up
//   ESCORTED  - follows you on foot; get into a car and he takes a passenger seat
//   BOARDING  - walking to the car / getting in
//   IN_CAR    - drive to the LSPD/DEA office (blip) and slow down there to book him:
//               the legal evidence bonus goes to the story balance
// The suspect is released when he dies, despawns or gets 60 m away.
{$CLEO .cs}
{$INCLUDE include\state.txt}
{$INCLUDE include\locations.txt}

script_name {name} 'deapolc'

const
    KEY_INTERACT = 0x54         // T
    FRISK_RANGE = 3.0
    CUFF_RANGE = 1.8
    CATCH_RANGE = 3.5
    RELEASE_RANGE = 60.0

    HANDS_UP_TIME = 2500        // ms with hands up before kneeling
    KNEEL_TIMEOUT = 60000       // ms a kneeling suspect waits to be cuffed
    CATCH_TIME = 1500           // ms you must stay close to a runner
    GIVE_UP_HEALTH = 50         // an armed suspect gives up below this

    // suspect states (outcomes 1-3 match the dev force keys 7/8/9)
    STATE_NONE = 0
    STATE_HANDS_UP = 1
    STATE_FLEEING = 2
    STATE_HOSTILE = 3
    STATE_KNEELING = 4
    STATE_CUFFED = 5
    STATE_ESCORTED = 6
    STATE_BOARDING = 7
    STATE_IN_CAR = 8

    MIRANDA_TIME = 3000         // ms on his knees after cuffing before he stands up
    BOARD_RANGE = 8.0           // he gets in if you enter a car this close to him
    BOARD_TIMEOUT = 12000
    BOOKING_RANGE = 15.0        // around the office door, covers the street in front
    BOOKING_MAX_SPEED = 3.0
    MAX_EVIDENCE = 4            // random 0..3 packages

    SURRENDER_CHANCE = 70       // roll 0-69
    FLEE_CHANCE_END = 90        // roll 70-89 flee, 90-99 fight

    PED_TYPE_COP = 6
    WEAPON_PISTOL = 22
    MODEL_PISTOL = 346
    BLIP_RED = 0
    BLIP_GREEN = 1
    BLIP_BLUE = 2
    SOUND_CUFFED = 1058         // frontend "part mission complete" sound

    // gta_sa.exe 1.0 US (plugin-sdk): void __thiscall CPed::SetCharCreatedBy(uchar)
    ADDR_SET_CHAR_CREATED_BY = 0x5E47E0
    PED_CREATED_BY_MISSION = 2
    DECISION_MAKER_EMPTY = 65542
end

int suspectHandle = -1
int suspectState = 0
int suspectBlip, nearbyCar, nearbyChar, nearbyPedType, diceRoll, forcedOutcome, suspectHealth
int releaseReason   // 0 = keep, 1 = lost, 2 = down, 3 = got away, 4 = left alone
int suspectPointer
int evidenceCount, bonusAmount, balanceNow, officeBlip, hasOfficeBlip, patrolCar, passengerSeat, maxPassengers
float carSpeed

wait {time} 2000

while true
    wait {time} 0

    if
        not is_player_playing $player1
    then
        continue
    end

    // ================= follow the current suspect =================
    if
        suspectState <> STATE_NONE
    then
        releaseReason = 0
        if
            not does_char_exist {handle} suspectHandle
        then
            releaseReason = 1
        else
            if
                is_char_dead {handle} suspectHandle
            then
                releaseReason = 2
            else
                if
                    not locate_char_any_means_char_2d {self} $scplayer {target} suspectHandle {radius} RELEASE_RANGE RELEASE_RANGE {drawSphere} false
                then
                    releaseReason = 3
                end
            end
        end

        if
            releaseReason == 0
        then
            // --- hands up -> kneel ---
            if
                suspectState == STATE_HANDS_UP
            then
                if
                    TimerA > HANDS_UP_TIME
                then
                    clear_char_tasks_immediately {self} suspectHandle
                    task_play_anim {handle} suspectHandle {animationName} "cower" {animationFile} "ped" {blendSpeed} 4.0 {loop} true {lockX} false {lockY} false {keepLastFrame} true {time} -1
                    set_char_keep_task {self} suspectHandle {state} true
                    suspectState = STATE_KNEELING
                    TimerA = 0
                end
            end

            // --- kneeling: T to cuff, or he leaves after a while ---
            if
                suspectState == STATE_KNEELING
            then
                if
                    is_key_pressed {keyCode} KEY_INTERACT
                then
                    if
                        locate_char_on_foot_char_2d {self} $scplayer {target} suspectHandle {radius} CUFF_RANGE CUFF_RANGE {drawSphere} false
                    then
                        suspectState = STATE_CUFFED
                        change_blip_colour {self} suspectBlip {color} BLIP_BLUE
                        add_one_off_sound {pos} 0.0 0.0 0.0 {soundId} SOUND_CUFFED
                        evidenceCount = generate_random_int_in_range {min} 0 {max} MAX_EVIDENCE
                        print_help_formatted {text} "~b~Suspect cuffed.~s~ Found on him: %d crack vial(s). Get him into a car and book him at the office." evidenceCount
                        print_string_now {text} "You have the right to remain silent. Anything you say can and will be used against you." {time} 4000
                        TimerA = 0
                    else
                        print_string_now {text} "~y~Get closer to cuff him" {time} 1500
                    end
                    while is_key_pressed {keyCode} KEY_INTERACT
                        wait {time} 0
                    end
                end
                if
                    suspectState == STATE_KNEELING
                then
                    if
                        TimerA > KNEEL_TIMEOUT
                    then
                        releaseReason = 4
                    end
                end
            end

            // --- runner: gives up once you stay close ---
            if
                suspectState == STATE_FLEEING
            then
                if
                    locate_char_any_means_char_2d {self} $scplayer {target} suspectHandle {radius} CATCH_RANGE CATCH_RANGE {drawSphere} false
                then
                    if
                        TimerB > CATCH_TIME
                    then
                        clear_char_tasks_immediately {self} suspectHandle
                        task_hands_up {handle} suspectHandle {time} HANDS_UP_TIME
                        change_blip_colour {self} suspectBlip {color} BLIP_GREEN
                        print_string_now {text} "~g~Okay, okay! Don't shoot! I give up!" {time} 2500
                        suspectState = STATE_HANDS_UP
                        TimerA = 0
                    end
                else
                    TimerB = 0
                end
            end

            // --- armed: gives up when badly hurt ---
            if
                suspectState == STATE_HOSTILE
            then
                suspectHealth = get_char_health {self} suspectHandle
                if
                    suspectHealth < GIVE_UP_HEALTH
                then
                    remove_all_char_weapons {self} suspectHandle
                    clear_char_tasks_immediately {self} suspectHandle
                    task_hands_up {handle} suspectHandle {time} HANDS_UP_TIME
                    change_blip_colour {self} suspectBlip {color} BLIP_GREEN
                    print_string_now {text} "~g~Don't shoot! I'm done, I'm done!" {time} 2500
                    suspectState = STATE_HANDS_UP
                    TimerA = 0
                end
            end

            // --- cuffed: after the Miranda rights he stands up and follows you ---
            if
                suspectState == STATE_CUFFED
            then
                if
                    TimerA > MIRANDA_TIME
                then
                    clear_char_tasks_immediately {self} suspectHandle
                    task_follow_footsteps {handle} suspectHandle {target} $scplayer
                    set_char_keep_task {self} suspectHandle {state} true
                    suspectState = STATE_ESCORTED
                end
            end

            // --- escorted: gets into your car when you get in close to him ---
            if
                suspectState == STATE_ESCORTED
            then
                if
                    is_char_sitting_in_any_car {self} $scplayer
                then
                    if
                        locate_char_any_means_char_2d {self} $scplayer {target} suspectHandle {radius} BOARD_RANGE BOARD_RANGE {drawSphere} false
                    then
                        patrolCar = store_car_char_is_in_no_save {self} $scplayer
                        maxPassengers = get_maximum_number_of_passengers {self} patrolCar
                        if
                            maxPassengers < 1
                        then
                            print_string_now {text} "~y~No room for him in this vehicle" {time} 1000
                        else
                            // prefer the back seats
                            passengerSeat = 0
                            if
                                maxPassengers > 1
                            then
                                if
                                    is_car_passenger_seat_free {self} patrolCar {seat} 1
                                then
                                    passengerSeat = 1
                                else
                                    if
                                        is_car_passenger_seat_free {self} patrolCar {seat} 2
                                    then
                                        passengerSeat = 2
                                    end
                                end
                            end
                            clear_char_tasks_immediately {self} suspectHandle
                            task_enter_car_as_passenger {char} suspectHandle {vehicle} patrolCar {time} BOARD_TIMEOUT {seat} passengerSeat
                            set_char_keep_task {self} suspectHandle {state} true
                            suspectState = STATE_BOARDING
                            TimerA = 0
                        end
                    end
                end
            end

            // --- the car is gone (destroyed or despawned): back to following you ---
            if
                suspectState >= STATE_BOARDING
            then
                if
                    not does_vehicle_exist {handle} patrolCar
                then
                    if
                        hasOfficeBlip == 1
                    then
                        remove_blip {self} officeBlip
                        hasOfficeBlip = 0
                    end
                    clear_char_tasks_immediately {self} suspectHandle
                    task_follow_footsteps {handle} suspectHandle {target} $scplayer
                    suspectState = STATE_ESCORTED
                end
            end

            // --- boarding: wait until he is seated; back to following if it fails ---
            if
                suspectState == STATE_BOARDING
            then
                if
                    is_char_sitting_in_car {self} suspectHandle {vehicle} patrolCar
                then
                    suspectState = STATE_IN_CAR
                    officeBlip = add_blip_for_coord {pos} OFFICE_EXT_X OFFICE_EXT_Y OFFICE_EXT_Z
                    hasOfficeBlip = 1
                    print_help_string {text} "Drive him to the ~b~LSPD/DEA office~s~ and stop in front of it to book him."
                else
                    if
                        not is_char_sitting_in_any_car {self} $scplayer
                    then
                        TimerA = BOARD_TIMEOUT
                    end
                    if
                        TimerA >= BOARD_TIMEOUT
                    then
                        clear_char_tasks_immediately {self} suspectHandle
                        task_follow_footsteps {handle} suspectHandle {target} $scplayer
                        suspectState = STATE_ESCORTED
                    end
                end
            end

            // --- in the car: book him at the office ---
            if
                suspectState == STATE_IN_CAR
            then
                if
                    not is_char_sitting_in_car {self} suspectHandle {vehicle} patrolCar
                then
                    // he left the car somehow: back to following you
                    remove_blip {self} officeBlip
                    hasOfficeBlip = 0
                    clear_char_tasks_immediately {self} suspectHandle
                    task_follow_footsteps {handle} suspectHandle {target} $scplayer
                    suspectState = STATE_ESCORTED
                else
                    if
                        locate_char_in_car_3d {self} $scplayer {pos} OFFICE_EXT_X OFFICE_EXT_Y OFFICE_EXT_Z {radius} BOOKING_RANGE BOOKING_RANGE 8.0 {drawSphere} false
                    then
                        carSpeed = get_car_speed {self} patrolCar
                        if
                            carSpeed < BOOKING_MAX_SPEED
                        then
                            do_fade {time} 500 {direction} 0
                            wait {time} 600
                            remove_blip {self} officeBlip
                            hasOfficeBlip = 0
                            remove_blip {self} suspectBlip
                            delete_char {self} suspectHandle
                            bonusAmount = evidenceCount * EVIDENCE_BONUS
                            balanceNow = get_cleo_shared_var {index} SV_BALANCE
                            balanceNow += bonusAmount
                            set_cleo_shared_var {index} SV_BALANCE {value} balanceNow
                            suspectState = STATE_NONE
                            wait {time} 400
                            do_fade {time} 500 {direction} 1
                            add_one_off_sound {pos} 0.0 0.0 0.0 {soundId} SOUND_CUFFED
                            print_help_formatted {text} "~g~Suspect booked.~s~ %d vial(s) logged as evidence. Legal bonus: ~g~$%d~s~" evidenceCount bonusAmount
                            continue
                        end
                    end
                end
            end
        end

        // --- release ---
        if
            releaseReason > 0
        then
            remove_blip {self} suspectBlip
            if
                hasOfficeBlip == 1
            then
                remove_blip {self} officeBlip
                hasOfficeBlip = 0
            end
            if
                releaseReason > 1
            then
                mark_char_as_no_longer_needed {self} suspectHandle
            end
            if
                releaseReason == 1
            then
                print_string_now {text} "~y~SUSPECT LOST" {time} 2000
            end
            if
                releaseReason == 2
            then
                print_string_now {text} "~r~SUSPECT DOWN" {time} 2000
            end
            if
                releaseReason == 3
            then
                print_string_now {text} "~y~SUSPECT GOT AWAY" {time} 2000
            end
            if
                releaseReason == 4
            then
                print_string_now {text} "~y~SUSPECT WALKED AWAY" {time} 2000
            end
            suspectState = STATE_NONE
        end
        continue
    end

    // ================= T: stop the closest pedestrian =================
    if
        is_key_pressed {keyCode} KEY_INTERACT
    then
        if
            is_char_on_foot {self} $scplayer
        then
            nearbyCar, nearbyChar = store_closest_entities {self} $scplayer
            if
                nearbyChar > 0
            then
                if
                    locate_char_on_foot_char_2d {self} $scplayer {target} nearbyChar {radius} FRISK_RANGE FRISK_RANGE {drawSphere} false
                then
                    if
                        not is_char_dead {handle} nearbyChar
                    then
                        nearbyPedType = get_ped_type {self} nearbyChar
                        if
                            nearbyPedType <> PED_TYPE_COP
                        then
                            suspectHandle = nearbyChar
                            // Take him over from the ambient AI: flag him as a mission ped (CPed::SetCharCreatedBy,
                            // so he no longer despawns or follows street logic like "leave a stranger's car and run")
                            // and give him the empty decision maker so he only reacts to our tasks.
                            suspectPointer = get_ped_pointer {char} suspectHandle
                            call_method {address} ADDR_SET_CHAR_CREATED_BY {struct} suspectPointer {numArgs} 1 {pop} 0 {args} PED_CREATED_BY_MISSION
                            set_char_decision_maker {self} suspectHandle {handleOrTemplate} DECISION_MAKER_EMPTY
                            print_string_now {text} "~b~DEA!~s~ Against the wall, hands where I can see them!" {time} 3000
                            clear_char_tasks_immediately {self} suspectHandle
                            task_turn_char_to_face_char {char} suspectHandle {target} $scplayer

                            // dice (dev keys 7/8/9 in dev_state_hud can force the next outcome)
                            forcedOutcome = get_cleo_shared_var {index} SV_DEV_FORCE_OUTCOME
                            if
                                forcedOutcome > STATE_NONE
                            then
                                suspectState = forcedOutcome
                                set_cleo_shared_var {index} SV_DEV_FORCE_OUTCOME {value} STATE_NONE
                            else
                                diceRoll = generate_random_int_in_range {min} 0 {max} 100
                                if
                                    diceRoll < SURRENDER_CHANCE
                                then
                                    suspectState = STATE_HANDS_UP
                                else
                                    if
                                        diceRoll < FLEE_CHANCE_END
                                    then
                                        suspectState = STATE_FLEEING
                                    else
                                        suspectState = STATE_HOSTILE
                                    end
                                end
                            end

                            suspectBlip = add_blip_for_char {char} suspectHandle
                            set_char_keep_task {self} suspectHandle {state} true

                            if
                                suspectState == STATE_HANDS_UP
                            then
                                task_hands_up {handle} suspectHandle {time} HANDS_UP_TIME
                                change_blip_colour {self} suspectBlip {color} BLIP_GREEN
                                print_string_now {text} "~g~SUSPECT SURRENDERS" {time} 2500
                                TimerA = 0
                            end
                            if
                                suspectState == STATE_FLEEING
                            then
                                task_smart_flee_char {handle} suspectHandle {threat} $scplayer {radius} 200.0 {time} -1
                                change_blip_colour {self} suspectBlip {color} BLIP_RED
                                print_string_now {text} "~r~SUSPECT IS RUNNING!" {time} 2500
                                TimerB = 0
                            end
                            if
                                suspectState == STATE_HOSTILE
                            then
                                request_model {modelId} MODEL_PISTOL
                                load_all_models_now
                                give_weapon_to_char {self} suspectHandle {weaponType} WEAPON_PISTOL {ammo} 60
                                set_current_char_weapon {self} suspectHandle {weaponType} WEAPON_PISTOL
                                mark_model_as_no_longer_needed {modelId} MODEL_PISTOL
                                set_char_accuracy {self} suspectHandle {accuracy} 35
                                task_kill_char_on_foot {killer} suspectHandle {target} $scplayer
                                change_blip_colour {self} suspectBlip {color} BLIP_RED
                                print_string_now {text} "~r~HE'S ARMED!" {time} 2500
                            end
                        end
                    end
                end
            end
        end
        while is_key_pressed {keyCode} KEY_INTERACT
            wait {time} 0
        end
    end
end
