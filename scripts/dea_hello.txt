// Installation check: press F7 in game to show "RAY CROSS - DEA".
{$CLEO .cs}

script_name {name} 'deahelo'

wait {time} 3000 // let the game finish loading

while true
    wait {time} 0

    if
        is_key_pressed {keyCode} 0x76 // F7
    then
        print_string_now {text} "RAY CROSS - DEA" {time} 3000

        // wait for key release, otherwise it would fire every frame
        while is_key_pressed {keyCode} 0x76
            wait {time} 0
        end
    end
end
